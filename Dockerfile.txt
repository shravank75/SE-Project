# Use an official Node.js runtime as the base image
FROM node:20-alpine AS base

# Set working directory
WORKDIR /app

# Copy package files
COPY package*.json ./

# Install dependencies
RUN npm install --production=false

# Copy rest of the app
COPY . .

# Build the Next.js app
RUN npm run build

# Expose port
EXPOSE 80

# Start the app
CMD ["npm", "run", "dev"]
